80.     private List<AiImageDO> selectByStatusAndPlatform() {
81.         List<AiImageDO> imageList = imageMapper.selectList(
82.                 new LambdaUpdateWrapper<AiImageDO>()
83.                         .eq(AiImageDO::getStatus, AiImageStatusEnum.IN_PROGRESS.getStatus())
84.                         .eq(AiImageDO::getPlatform, AiPlatformEnum.MIDJOURNEY.getPlatform())
85.         );
86.         return imageList;
