107.     private CrmContractRespVO buildContractDetail(CrmContractDO contract) {
108.         if (contract == null) {
109.             return null;
110.         }
111.         CrmContractRespVO contractVO = buildContractDetailList(singletonList(contract)).get(0);
112.         // 拼接产品项
113.         List<CrmContractProductDO> businessProducts = contractService.getContractProductListByContractId(contractVO.getId());
114.         Map<Long, CrmProductDO> productMap = productService.getProductMap(
115.                 convertSet(businessProducts, CrmContractProductDO::getProductId));
116.         contractVO.setProducts(BeanUtils.toBean(businessProducts, CrmContractRespVO.Product.class, businessProductVO ->
117.                 MapUtils.findAndThen(productMap, businessProductVO.getProductId(),
118.                         product -> businessProductVO.setProductName(product.getName())
119.                                 .setProductNo(product.getNo()).setProductUnit(product.getUnit()))));
120.         return contractVO;
121.     }

167.     /**
168.      * 构建详细的合同结果
169.      *
170.      * @param contractList 原始合同信息
171.      * @return 细的合同结果
172.      */
173.     private List<CrmContractRespVO> buildContractDetailList(List<CrmContractDO> contractList) {
174.         if (CollUtil.isEmpty(contractList)) {
175.             return Collections.emptyList();
176.         }
177.         // 1.1 获取客户列表
178.         Map<Long, CrmCustomerDO> customerMap = customerService.getCustomerMap(
179.                 convertSet(contractList, CrmContractDO::getCustomerId));
180.         // 1.2 获取创建人、负责人列表
181.         Map<Long, AdminUserRespDTO> userMap = adminUserApi.getUserMap(convertListByFlatMap(contractList,
182.                 contact -> Stream.of(NumberUtils.parseLong(contact.getCreator()), contact.getOwnerUserId())));
183.         Map<Long, DeptRespDTO> deptMap = deptApi.getDeptMap(convertSet(userMap.values(), AdminUserRespDTO::getDeptId));
184.         // 1.3 获取联系人
185.         Map<Long, CrmContactDO> contactMap = convertMap(contactService.getContactList(convertSet(contractList,
186.                 CrmContractDO::getSignContactId)), CrmContactDO::getId);
187.         // 1.4 获取商机
188.         Map<Long, CrmBusinessDO> businessMap = getBusinessMap(convertSet(contractList,
189.                 CrmContractDO::getBusinessId));
190.         // 2. 拼接数据
191.         return BeanUtils.toBean(contractList, CrmContractRespVO.class, contractVO -> {
192.             // 2.1 设置客户信息
193.             findAndThen(customerMap, contractVO.getCustomerId(), customer -> contractVO.setCustomerName(customer.getName()));
194.             // 2.2 设置用户信息
195.             findAndThen(userMap, Long.parseLong(contractVO.getCreator()), user -> contractVO.setCreatorName(user.getNickname()));
196.             MapUtils.findAndThen(userMap, contractVO.getOwnerUserId(), user -> {
197.                 contractVO.setOwnerUserName(user.getNickname());
198.                 MapUtils.findAndThen(deptMap, user.getDeptId(), dept -> contractVO.setOwnerUserDeptName(dept.getName()));
199.             });
200.             findAndThen(userMap, contractVO.getSignUserId(), user -> contractVO.setSignUserName(user.getNickname()));
201.             // 2.3 设置联系人信息
202.             findAndThen(contactMap, contractVO.getSignContactId(), contact -> contractVO.setSignContactName(contact.getName()));
203.             // 2.4 设置商机信息
204.             findAndThen(businessMap, contractVO.getBusinessId(), business -> contractVO.setBusinessName(business.getName()));
205.         });
206.     }
207. 
208.     private Map<Long, CrmBusinessDO> getBusinessMap(Collection<Long> ids) {
209.         return convertMap(businessService.getBusinessList(ids), CrmBusinessDO::getId);
210.     }
