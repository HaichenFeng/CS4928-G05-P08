44.     @Override
45.     @Transactional(rollbackFor = Exception.class)
46.     public Long createBusinessStatusType(CrmBusinessStatusTypeSaveReqVO createReqVO) {
47.         //检验名称是否存在
48.         validateBusinessStatusTypeExists(createReqVO.getName(), null);
49.         // 插入类型
50.         CrmBusinessStatusTypeDO businessStatusType = BeanUtils.toBean(createReqVO, CrmBusinessStatusTypeDO.class);
51.         businessStatusTypeMapper.insert(businessStatusType);
52.         // 插入状态
53.         if (CollUtil.isNotEmpty(createReqVO.getStatusList())) {
54.             createReqVO.getStatusList().forEach(status -> status.setTypeId(businessStatusType.getId()));
55.             businessStatusMapper.insertBatch(BeanUtils.toBean(createReqVO.getStatusList(), CrmBusinessStatusDO.class));
56.         }
57.         return businessStatusType.getId();
58.     }
59.

62.     public void updateBusinessStatusType(CrmBusinessStatusTypeSaveReqVO updateReqVO) {
63.         // 校验存在
64.         validateBusinessStatusTypeExists(updateReqVO.getId());
65.         // 校验名称是否存在
66.         validateBusinessStatusTypeExists(updateReqVO.getName(), updateReqVO.getId());
67.         // 更新类型
68.         CrmBusinessStatusTypeDO updateObj = BeanUtils.toBean(updateReqVO, CrmBusinessStatusTypeDO.class);
69.         businessStatusTypeMapper.updateById(updateObj);
70.         // 更新状态（删除 + 更新）
71.         // TODO @ljlleo 可以参考 DeliveryExpressTemplateServiceImpl 的 updateExpressTemplateFree 方法；主要没变化的，还是不删除了哈。
72.         businessStatusMapper.delete(updateReqVO.getId());
73.         updateReqVO.getStatusList().forEach(status -> status.setTypeId(updateReqVO.getId()));
74.         businessStatusMapper.insertBatch(BeanUtils.toBean(updateReqVO.getStatusList(), CrmBusinessStatusDO.class));
75.     }

97.     // TODO @ljlleo 这个方法，这个参考 validateDeptNameUnique 实现。
98.     private void validateBusinessStatusTypeExists(String name, Long id) {
99.         LambdaQueryWrapper<CrmBusinessStatusTypeDO> wrapper = new LambdaQueryWrapperX<>();
100.         if(null != id) {
101.             wrapper.ne(selectByIdAndName(), id);
102.         }
103.         long cnt = businessStatusTypeMapper.selectCount(wrapper.eq(CrmBusinessStatusTypeDO::getName, name));
104.         if (cnt > 0) {
105.             throw exception(BUSINESS_STATUS_TYPE_NAME_EXISTS);
106.         }
107.     }
